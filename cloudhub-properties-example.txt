#
# Produção (Baixo Tráfego):
# - Worker Size: 0.2 vCores (SMALL)
# - Workers: 1
# - Region: us-east-2
#
# Produção (Médio Tráfego):
# - Worker Size: 1 vCore (MEDIUM)
# - Workers: 2 (para alta disponibilidade)
# - Region: us-east-2
#
# Produção (Alto Tráfego):
# - Worker Size: 2 vCores (LARGE)
# - Workers: 2+
# - Region: us-east-2
